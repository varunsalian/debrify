import 'dart:async';
class _VideoPlayerScreenState {
  int _x = 0; // keep the initializer comment
  int _y = 1;
  bool get mounted => true;
  Object get context => Object();
  void setState(void Function() fn) => fn();
  void _vpSetState(void Function() fn) => setState(fn);
  void retain() { _x += 1; }
  static String _pad2(int n) => n.toString().padLeft(2, '0');
  void move(int x) {
    // preserve this comment
    final snapshot = _x;
    _x = x;
    _x += 2;
    _x++;
    use(() => snapshot);
    owner(this);
    if (!mounted) return;
    use(context);
    setState(() { _x = 3; });
    builder((context, setSheetState) { use(context); setSheetState(() {}); });
    builder((context, setDialogState) { setDialogState(() {}); });
  }
}
void owner(Object x) {}
void use(Object x) {}
void builder(void Function(Object, void Function(void Function())) x) {}
