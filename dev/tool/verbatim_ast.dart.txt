// AST/token adapter for check_verbatim_moves.py. Tool-only analyzer 14.3.0;
// its isolated package config never changes the application's dependencies.
import 'dart:convert';
import 'dart:io';
import 'package:analyzer/dart/analysis/utilities.dart';
import 'package:analyzer/dart/ast/ast.dart';
import 'package:analyzer/dart/ast/visitor.dart';
import 'package:analyzer/dart/ast/token.dart';

class Identifiers extends RecursiveAstVisitor<void> {
  final List<Map<String, Object>> values = [];
  @override
  void visitSimpleIdentifier(SimpleIdentifier node) {
    values.add({'name': node.name, 'offset': node.offset, 'end': node.end});
    super.visitSimpleIdentifier(node);
  }
}

// Additional evidence for the Settings extraction. This is syntax analysis,
// not type resolution: consumers restrict references to the declared owner.
class References extends RecursiveAstVisitor<void> {
  final List<Map<String, Object?>> values = [];
  final List<Map<String, Object?>> callbacks = [];
  final List<String> scopes = [];
  final List<Set<String>> parameters = [];
  final List<int> formatCommas = [];
  final List<Map<String, Object?>> namedRecords = [];
  final List<Map<String, Object?>> awaits = [];

  @override
  void visitAwaitExpression(AwaitExpression node) {
    awaits.add({
      'offset': node.offset,
      'end': node.end,
      'scope': scopes.join('/'),
      'contextParameter': parameters.any((p) => p.contains('context')),
    });
    super.visitAwaitExpression(node);
  }

  @override
  void visitArgumentList(ArgumentList node) {
    if (node.endToken.previous?.lexeme == ',') {
      formatCommas.add(node.endToken.previous!.offset);
    }
    super.visitArgumentList(node);
  }

  @override
  void visitFormalParameterList(FormalParameterList node) {
    if (node.endToken.previous?.lexeme == ',') {
      formatCommas.add(node.endToken.previous!.offset);
    }
    // Named/optional parameters close with } or ] before the final ). Their
    // trailing comma is formatting punctuation too, never a parameter separator.
    final optionalEnd = node.endToken.previous;
    if ((optionalEnd?.lexeme == '}' || optionalEnd?.lexeme == ']') &&
        optionalEnd?.previous?.lexeme == ',') {
      formatCommas.add(optionalEnd!.previous!.offset);
    }
    super.visitFormalParameterList(node);
  }

  @override
  void visitRecordLiteral(RecordLiteral node) {
    // The comma in a one-positional-field record distinguishes it from a
    // parenthesized expression. Only multi-field positional records have an
    // optional formatter comma here; named dependency records stay pinned.
    if (node.fields.length > 1 &&
        node.fields.every((field) => field is! RecordLiteralNamedField) &&
        node.endToken.previous?.lexeme == ',') {
      formatCommas.add(node.endToken.previous!.offset);
    }
    if (node.fields.every((field) => field is RecordLiteralNamedField)) {
      namedRecords.add({
        'offset': node.offset,
        'end': node.end,
        'trailingComma': node.endToken.previous?.lexeme == ','
            ? node.endToken.previous!.offset
            : null,
      });
    }
    super.visitRecordLiteral(node);
  }

  @override
  void visitRecordTypeAnnotation(RecordTypeAnnotation node) {
    final named = node.namedFields;
    if (named?.endToken.previous?.lexeme == ',') {
      formatCommas.add(named!.endToken.previous!.offset);
    }
    super.visitRecordTypeAnnotation(node);
  }

  @override
  void visitSetOrMapLiteral(SetOrMapLiteral node) {
    if (node.endToken.previous?.lexeme == ',') {
      formatCommas.add(node.endToken.previous!.offset);
    }
    super.visitSetOrMapLiteral(node);
  }

  @override
  void visitListLiteral(ListLiteral node) {
    if (node.endToken.previous?.lexeme == ',') {
      formatCommas.add(node.endToken.previous!.offset);
    }
    super.visitListLiteral(node);
  }

  @override
  void visitMethodDeclaration(MethodDeclaration node) {
    // Method parameters also bind nested callbacks, just like function ones.
    parameters.add({
      for (final p in node.parameters?.parameters ?? <FormalParameter>[])
        p.name!.lexeme,
    });
    super.visitMethodDeclaration(node);
    parameters.removeLast();
  }

  @override
  void visitFunctionExpression(FunctionExpression node) {
    scopes.add('${callbacks.length}');
    parameters.add({
      for (final p in node.parameters?.parameters ?? <FormalParameter>[])
        p.name!.lexeme,
    });
    callbacks.add({
      'offset': node.offset,
      'end': node.end,
      'scope': scopes.join('/'),
      'parent': node.parent.runtimeType.toString(),
      'parameters': parameters.last.toList(),
    });
    super.visitFunctionExpression(node);
    parameters.removeLast();
    scopes.removeLast();
  }

  @override
  void visitSimpleIdentifier(SimpleIdentifier node) {
    final parent = node.parent;
    if (parent is Label || node.inDeclarationContext()) return;
    final invoked = parent is MethodInvocation && parent.methodName == node;
    final assigned =
        parent is AssignmentExpression && parent.leftHandSide == node;
    final updated =
        parent is PostfixExpression ||
        parent is PrefixExpression &&
            (parent.operator.lexeme == '++' || parent.operator.lexeme == '--');
    AstNode? statement = parent;
    while (statement != null && statement is! Statement) {
      statement = statement.parent;
    }
    final blocks = <int>[];
    for (
      AstNode? ancestor = parent;
      ancestor != null;
      ancestor = ancestor.parent
    ) {
      if (ancestor is Block) blocks.add(ancestor.offset);
    }
    final lexicalNames = <String>{
      for (final names in parameters) ...names,
    };
    for (AstNode? ancestor = parent; ancestor != null; ancestor = ancestor.parent) {
      if (ancestor is Block) {
        for (final statement in ancestor.statements) {
          if (statement is VariableDeclarationStatement) {
            for (final variable in statement.variables.variables) {
              lexicalNames.add(variable.name.lexeme);
            }
          }
        }
      }
    }
    final receiver = switch (parent) {
      PropertyAccess() when parent.propertyName == node => parent.target?.toSource(),
      PrefixedIdentifier() when parent.identifier == node => parent.prefix.toSource(),
      MethodInvocation() when parent.methodName == node => parent.target?.toSource(),
      _ => null,
    };
    final bindingNode = switch (parent) {
      PropertyAccess() when parent.propertyName == node => parent,
      PrefixedIdentifier() when parent.identifier == node => parent,
      _ => node,
    };
    final bindingParent = bindingNode.parent;
    final bindingRole = invoked ? 'call'
        : bindingParent is AssignmentExpression && bindingParent.leftHandSide == bindingNode
        ? (bindingParent.operator.lexeme == '=' ? 'write' : 'read/write')
        : bindingParent is PostfixExpression || bindingParent is PrefixExpression &&
          (bindingParent.operator.lexeme == '++' || bindingParent.operator.lexeme == '--')
        ? 'read/write' : 'read';
    values.add({
      'bindingRole': bindingRole,
      'lexicalNames': lexicalNames.toList(),
      'receiver': receiver,
      'name': node.name,
      'offset': node.offset,
      'end': node.end,
      'role': invoked
          ? 'call'
          : assigned
          ? (parent.operator.lexeme == '=' ? 'write' : 'read/write')
          : updated
          ? 'read/write'
          : 'read',
      'scope': scopes.join('/'),
      'contextParameter': parameters.any((p) => p.contains('context')),
      'parent': parent.runtimeType.toString(),
      'statementOffset': statement?.offset,
      'statementInBlock': statement?.parent is Block,
      'blocks': blocks,
    });
    super.visitSimpleIdentifier(node);
  }
}

// Opt-in occurrence evidence. Two passes collect declarations before references,
// preserving lexical binding and selected-member qualifiers before normalization.
class _BindingScope {
  final AstNode node;
  final String path;
  final _BindingScope? parent;
  final Map<String, List<(int, String)>> declarations = {};
  final Map<String, int> children = {};
  _BindingScope(this.node, this.path, this.parent);
  void declare(String name, int offset) {
    final entries = declarations.putIfAbsent(name, () => []);
    entries.add((offset, 'decl:$path#$name#${entries.length}'));
  }
}

class OccurrenceBindings {
  final AstNode root;
  final Map<AstNode, _BindingScope> scopes = {};
  final Map<AstNode, _BindingScope> enclosing = {};
  final List<Map<String, Object?>> values = [];
  final Map<int, int> tokenIndices;
  final int Function(int) position;
  OccurrenceBindings(this.root, this.tokenIndices, this.position) {
    final scope = _BindingScope(root, 'member', null);
    scopes[root] = scope;
    _collect(root, scope);
    _references(root);
    values.sort((a, b) => (a['tokenIndex'] as int).compareTo(b['tokenIndex'] as int));
  }
  String? _scopeKind(AstNode node) => switch (node) {
    MethodDeclaration() || ConstructorDeclaration() => 'method',
    FunctionDeclaration() => 'function',
    FunctionExpression() => 'closure',
    Block() => 'block',
    ForStatement() || ForElement() => 'for',
    CatchClause() => 'catch',
    IfStatement() || IfElement() => 'if',
    SwitchCase() || SwitchPatternCase() || SwitchExpressionCase() => 'pattern',
    _ => null,
  };
  void _collect(AstNode node, _BindingScope scope) {
    // A local function's name belongs to the surrounding scope.
    if (node is FunctionDeclaration && node != root && node.parent is FunctionDeclarationStatement) {
      scope.declare(node.name.lexeme, node.name.offset);
    }
    final kind = node == root ? null : _scopeKind(node);
    if (kind != null) {
      final ordinal = scope.children.update(kind, (n) => n + 1, ifAbsent: () => 0);
      scope = _BindingScope(node, '${scope.path}/$kind$ordinal', scope);
      scopes[node] = scope;
    }
    enclosing[node] = scope;
    if (node is FormalParameter && node.name != null) {
      scope.declare(node.name!.lexeme, node.name!.offset);
    } else if (node is VariableDeclaration && node.parent?.parent is! FieldDeclaration && node.parent?.parent is! TopLevelVariableDeclaration) {
      scope.declare(node.name.lexeme, node.name.offset);
    } else if (node is DeclaredIdentifier) {
      scope.declare(node.name.lexeme, node.name.offset);
    } else if (node is DeclaredVariablePattern) {
      scope.declare(node.name.lexeme, node.name.offset);
    } else if (node is CatchClauseParameter) {
      scope.declare(node.name.lexeme, node.name.offset);
    }
    for (final child in node.childEntities.whereType<AstNode>()) {
      _collect(child, scope);
    }
  }
  String _binding(SimpleIdentifier node, _BindingScope scope) {
    var delayed = false;
    for (_BindingScope? current = scope; current != null; current = current.parent) {
      final entries = current.declarations[node.name];
      if (entries != null) {
        final (offset, identity) = entries.first;
        return node.offset < offset && !delayed ? 'early:$identity' : identity;
      }
      if (current.node is FunctionExpression || current.node is FunctionDeclaration) {
        delayed = true;
      }
    }
    return 'unbound';
  }
  List<String> _qualifierTokens(Expression target) {
    final result = <String>[];
    final formatting = References();
    target.accept(formatting);
    final optionalCommas = formatting.formatCommas.toSet();
    var token = target.beginToken;
    while (true) {
      if (!optionalCommas.contains(token.offset)) result.add(token.lexeme);
      if (token == target.endToken || token.next == null) break;
      token = token.next!;
    }
    return result;
  }
  void _references(AstNode node) {
    if (node is Comment || node is CommentReference) return;
    if (node is SimpleIdentifier && !node.inDeclarationContext() && node.parent is! Label) {
      final parent = node.parent;
      Expression? target;
      if (parent is PropertyAccess && parent.propertyName == node) target = parent.realTarget;
      if (parent is PrefixedIdentifier && parent.identifier == node) target = parent.prefix;
      if (parent is MethodInvocation && parent.methodName == node) target = parent.realTarget;
      final selected = target != null;
      final bindingNode = selected && (parent is PropertyAccess || parent is PrefixedIdentifier) ? parent! : node;
      final bindingParent = bindingNode.parent;
      final role = parent is MethodInvocation && parent.methodName == node ? 'invocation'
          : bindingParent is AssignmentExpression && bindingParent.leftHandSide == bindingNode
              ? (bindingParent.operator.lexeme == '=' ? 'write' : 'read-write')
          : bindingParent is PostfixExpression || bindingParent is PrefixExpression &&
              (bindingParent.operator.lexeme == '++' || bindingParent.operator.lexeme == '--')
              ? 'read-write' : 'read';
      final scope = enclosing[node]!;
      values.add({
        'name': node.name,
        'tokenIndex': tokenIndices[position(node.offset)],
        'offset': position(node.offset),
        'end': position(node.end),
        'scope': scope.path,
        'kind': selected ? 'selector' : 'lexical',
        'qualifier': target?.toSource(),
        'qualifierTokens': target == null ? null : _qualifierTokens(target),
        'binding': selected ? null : _binding(node, scope),
        'role': role,
      });
    }
    for (final child in node.childEntities.whereType<AstNode>()) {
      _references(child);
    }
  }
}

Map<String, Object?> listEvidence(
  ListLiteral list,
  String source,
  List<int> positions,
) => {
  'offset': positions[list.offset],
  'end': positions[list.end],
  'elements': [
    for (final e in list.elements)
      {
        'kind': e.runtimeType.toString(),
        'offset': positions[e.offset],
        'end': positions[e.end],
        'text': source.substring(e.offset, e.end),
        'line': '\n'.allMatches(source.substring(0, e.offset)).length + 1,
        'endLine': '\n'.allMatches(source.substring(0, e.end)).length + 1,
      },
  ],
};

// Inspect only the owning function's block. A returned list in a callback is
// not the builder's result, and collection if/for nodes remain whole elements.
Map<String, Object?>? returnedList(
  FunctionBody? body,
  String source,
  List<int> positions,
) {
  if (body is! BlockFunctionBody) return null;
  for (final statement in body.block.statements) {
    if (statement is ReturnStatement && statement.expression is ListLiteral) {
      return {
        ...listEvidence(statement.expression as ListLiteral, source, positions),
        'statementOffset': positions[statement.offset],
        'preamble': [
          for (final s in body.block.statements.takeWhile(
            (s) => s != statement,
          ))
            {
              'kind': s.runtimeType.toString(),
              'offset': positions[s.offset],
              'end': positions[s.end],
              'text': source.substring(s.offset, s.end),
              'names': s is VariableDeclarationStatement
                  ? [for (final v in s.variables.variables) v.name.lexeme]
                  : s is FunctionDeclarationStatement
                  ? [s.functionDeclaration.name.lexeme]
                  : [],
            },
        ],
      };
    }
  }
  return null;
}

Map<String, Object?> parameterDetails(FormalParameter parameter) {
  return {
    'name': parameter.name?.lexeme,
    'type': parameter.type?.toSource(),
    'syntax': parameter.toSource(),
    'functionTypedSuffix': parameter.functionTypedSuffix?.toSource(),
    'named': parameter.isNamed,
    'requiredNamed': parameter.isRequiredNamed,
    'optionalPositional': parameter.isOptionalPositional,
    'default': parameter.defaultClause?.value2.toSource(),
  };
}

String nameOf(AstNode node) {
  if (node is ClassDeclaration) return node.namePart.typeName.lexeme;
  if (node is EnumDeclaration) return node.namePart.typeName.lexeme;
  if (node is ExtensionDeclaration) return node.name?.lexeme ?? '<unnamed>';
  if (node is FunctionDeclaration) return node.name.lexeme;
  if (node is GenericTypeAlias) return node.name.lexeme;
  if (node is MethodDeclaration) return node.name.lexeme;
  if (node is ConstructorDeclaration) {
    return node.typeName2!.lexeme +
        (node.name == null ? '' : '.${node.name!.lexeme}');
  }
  if (node is FieldDeclaration) return node.fields.variables.single.name.lexeme;
  if (node is TopLevelVariableDeclaration)
    return node.variables.variables.single.name.lexeme;
  throw StateError(
    'Unsupported declaration ${node.runtimeType}: ${node.toSource()}',
  );
}

Map<String, Object?> describe(
  AstNode node,
  String source, {
  String owner = '',
  bool occurrenceBindings = false,
  required List<int> positions,
}) {
  int pos(int offset) => positions[offset];
  final ids = Identifiers();
  node.accept(ids);
  final refs = References();
  node.accept(refs);
  final start = node is AnnotatedNode
      ? node.documentationComment?.offset ?? node.offset
      : node.offset;
  final tokens = <Map<String, Object>>[];
  var token = node is AnnotatedNode
      ? (node.metadata.isEmpty
            ? node.firstTokenAfterCommentAndMetadata
            : node.metadata.first.beginToken)
      : node.beginToken;
  while (true) {
    Token? comment = token.precedingComments;
    while (comment != null) {
      if (comment.offset >= start) {
        tokens.add({
          'text': comment.lexeme,
          'offset': pos(comment.offset),
          'comment': true,
        });
      }
      comment = comment.next;
    }
    tokens.add({
      'text': token.lexeme,
      'offset': pos(token.offset),
      'comment': false,
    });
    if (token.offset >= node.endToken.offset || token.next == null) break;
    token = token.next!;
  }
  final members = node is ClassDeclaration
      ? node.body.members
      : node is ExtensionDeclaration
      ? node.body.members
      : <ClassMember>[];
  FormalParameterList? parameters;
  FunctionBody? body;
  if (node is MethodDeclaration) {
    parameters = node.parameters;
    body = node.body;
  }
  if (node is FunctionDeclaration) {
    parameters = node.functionExpression.parameters;
    body = node.functionExpression.body;
  }
  if (node is ConstructorDeclaration) parameters = node.parameters;
  final variableType = node is FieldDeclaration
      ? node.fields.type
      : node is TopLevelVariableDeclaration
      ? node.variables.type
      : null;
  return {
    'name': nameOf(node),
    'owner': owner,
    'kind': node.runtimeType.toString(),
    'offset': pos(node.offset),
    'start': pos(start),
    'end': pos(node.end),
    'text': source.substring(start, node.end),
    'tokens': tokens,
    'identifiers': [
      for (final id in ids.values)
        {
          ...id,
          'offset': pos(id['offset'] as int),
          'end': pos(id['end'] as int),
        },
    ],
    'members': [
      for (final m in members)
        describe(m, source, owner: nameOf(node), positions: positions, occurrenceBindings: occurrenceBindings),
    ],
    'endTokenText': node.endToken.lexeme,
    'parameters': parameters?.toSource(),
    'returnType': node is MethodDeclaration
        ? node.returnType?.toSource()
        : node is FunctionDeclaration
        ? node.returnType?.toSource()
        : null,
    'parameterDetails': [
      for (final p in parameters?.parameters ?? <FormalParameter>[])
        parameterDetails(p),
    ],
    'parameterNames': [
      for (final p in parameters?.parameters ?? <FormalParameter>[])
        p.name?.lexeme,
    ],
    'namedParameters': [
      for (final p in parameters?.parameters ?? <FormalParameter>[]) p.isNamed,
    ],
    'bodyOffset': body == null ? null : pos(body.offset),
    'bodyText': body?.toSource(),
    'recordTypeFields': variableType is RecordTypeAnnotation
        ? [
            for (final field
                in variableType.namedFields?.fields ??
                    <RecordTypeAnnotationNamedField>[])
              field.name.lexeme,
          ]
        : null,
    'variables': node is FieldDeclaration || node is TopLevelVariableDeclaration
        ? [
            for (final v
                in (node is FieldDeclaration
                        ? node.fields
                        : (node as TopLevelVariableDeclaration).variables)
                    .variables)
              {
                'name': v.name.lexeme,
                'initializerOffset': v.initializer == null
                    ? null
                    : pos(v.initializer!.offset),
                'initializerEnd': v.initializer == null
                    ? null
                    : pos(v.initializer!.end),
              },
          ]
        : null,
    'initializers': node is ConstructorDeclaration
        ? [
            for (final i in node.initializers)
              {
                'offset': pos(i.offset),
                'end': pos(i.end),
                'name': i is ConstructorFieldInitializer
                    ? i.fieldName.name
                    : null,
                'expressionOffset': i is ConstructorFieldInitializer
                    ? pos(i.expression.offset)
                    : null,
                'expressionEnd': i is ConstructorFieldInitializer
                    ? pos(i.expression.end)
                    : null,
              },
          ]
        : null,
    'references': [
      for (final r in refs.values)
        {
          ...r,
          'offset': pos(r['offset'] as int),
          'end': pos(r['end'] as int),
          'statementOffset': r['statementOffset'] == null
              ? null
              : pos(r['statementOffset'] as int),
          'blocks': [for (final block in r['blocks'] as List<int>) pos(block)],
        },
    ],
    'formatCommas': [for (final offset in refs.formatCommas) pos(offset)],
    'namedRecords': [
      for (final record in refs.namedRecords)
        {
          'offset': pos(record['offset'] as int),
          'end': pos(record['end'] as int),
          'trailingComma': record['trailingComma'] == null
              ? null
              : pos(record['trailingComma'] as int),
        },
    ],
    'awaits': [
      for (final a in refs.awaits)
        {...a, 'offset': pos(a['offset'] as int), 'end': pos(a['end'] as int)},
    ],
    'callbacks': [
      for (final c in refs.callbacks)
        {...c, 'offset': pos(c['offset'] as int), 'end': pos(c['end'] as int)},
    ],
    if (occurrenceBindings) 'occurrences': OccurrenceBindings(node, {for (var i = 0; i < tokens.length; i++) tokens[i]['offset'] as int: i}, pos).values,
    'returnedList': returnedList(body, source, positions),
  };
}

void main() {
  final envelope = jsonDecode(stdin.readLineSync()!) as Map<String, dynamic>;
  final occurrenceBindings = envelope[r'$occurrence_field_bindings'] == true;
  final input = occurrenceBindings ? envelope['sources'] as Map<String, dynamic> : envelope;
  final output = <String, Object>{};
  for (final entry in input.entries) {
    final source = entry.value as String;
    final parsed = parseString(content: source, throwIfDiagnostics: false);
    if (parsed.errors.isNotEmpty) {
      throw StateError('${entry.key}: ${parsed.errors}');
    }
    final positions = <int>[0];
    var point = 0;
    for (final rune in source.runes) {
      if (rune > 0xffff) positions.add(point);
      positions.add(++point);
    }
    output[entry.key] = {
      'declarations': [
        for (final d in parsed.unit.declarations)
          describe(d, source, positions: positions, occurrenceBindings: occurrenceBindings),
      ],
      'directives': [for (final d in parsed.unit.directives) d.toSource()],
    };
  }
  stdout.write(jsonEncode(output));
}
