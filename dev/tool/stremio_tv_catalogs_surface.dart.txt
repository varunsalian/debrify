import 'dart:convert';
import 'package:analyzer/dart/analysis/analysis_context_collection.dart';
import 'package:analyzer/dart/analysis/results.dart';

Future<void> main(List<String> args) async {
  final collection = AnalysisContextCollection(includedPaths: [args.single]);
  try {
    final context = collection.contextFor(args.single);
    final result = await context.currentSession.getLibraryByUri(Uri.file(args.single).toString());
    if (result is! LibraryElementResult) throw StateError('Library did not resolve: $result');
    final names = result.element.exportNamespace.definedNames2.keys.toList()..sort();
    print(jsonEncode(names));
  } finally { await collection.dispose(); }
}
